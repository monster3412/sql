
DROP DATABASE IF EXISTS coffee_shop_bi;
CREATE DATABASE coffee_shop_bi CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE coffee_shop_bi;

CREATE TABLE points (
    point_id VARCHAR(3) PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    address VARCHAR(100) NOT NULL,
    license_status VARCHAR(10) DEFAULT 'Активна'
);

CREATE TABLE categories (
    cat_id VARCHAR(3) PRIMARY KEY,
    name VARCHAR(50) NOT NULL
);

CREATE TABLE suppliers (
    sup_id VARCHAR(3) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL
);

CREATE TABLE employees (
    emp_id VARCHAR(6) PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    role VARCHAR(20) NOT NULL,
    login VARCHAR(50) UNIQUE NOT NULL,
    pass_hash VARCHAR(64) NOT NULL
);


CREATE TABLE storage_locations (
    loc_id VARCHAR(4) PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    point_id VARCHAR(3) NOT NULL,
    FOREIGN KEY (point_id) REFERENCES points(point_id) ON DELETE RESTRICT
);

CREATE TABLE products (
    prod_id VARCHAR(6) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    unit VARCHAR(10) NOT NULL,
    cat_id VARCHAR(3) NOT NULL,
    FOREIGN KEY (cat_id) REFERENCES categories(cat_id) ON DELETE RESTRICT
);

CREATE TABLE shifts (
    shift_id VARCHAR(8) PRIMARY KEY,
    point_id VARCHAR(3) NOT NULL,
    open_date DATETIME NOT NULL,
    close_date DATETIME,
    status VARCHAR(10) DEFAULT 'Активна',
    cash_balance DECIMAL(12,2) DEFAULT 0.00,
    shift_target DECIMAL(12,2) DEFAULT 0.00,
    tips_sum DECIMAL(12,2) DEFAULT 0.00,
    FOREIGN KEY (point_id) REFERENCES points(point_id) ON DELETE RESTRICT
);

CREATE TABLE menu (
    menu_id VARCHAR(6) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    price DECIMAL(10,2) NOT NULL
);

-- Связь многие-ко-многим: меню ↔ товары (рецептура)
CREATE TABLE recipes (
    menu_id VARCHAR(6),
    prod_id VARCHAR(6),
    qty DECIMAL(10,3) NOT NULL,
    PRIMARY KEY (menu_id, prod_id),
    FOREIGN KEY (menu_id) REFERENCES menu(menu_id) ON DELETE CASCADE,
    FOREIGN KEY (prod_id) REFERENCES products(prod_id) ON DELETE RESTRICT
);


CREATE TABLE warehouse_ops (
    op_id VARCHAR(10) PRIMARY KEY,
    shift_id VARCHAR(8) NOT NULL,
    prod_id VARCHAR(6) NOT NULL,
    loc_id VARCHAR(4) NOT NULL,
    menu_id VARCHAR(6),
    emp_id VARCHAR(6) NOT NULL,
    op_date DATE NOT NULL,
    op_type VARCHAR(30) NOT NULL,
    op_reason VARCHAR(30),
    comment VARCHAR(200),
    qty DECIMAL(10,3) NOT NULL,
    price_per_unit DECIMAL(10,2),
    sup_id VARCHAR(3),
    expiry_date DATE,
    FOREIGN KEY (shift_id) REFERENCES shifts(shift_id) ON DELETE RESTRICT,
    FOREIGN KEY (prod_id) REFERENCES products(prod_id) ON DELETE RESTRICT,
    FOREIGN KEY (loc_id) REFERENCES storage_locations(loc_id) ON DELETE RESTRICT,
    FOREIGN KEY (menu_id) REFERENCES menu(menu_id) ON DELETE RESTRICT,
    FOREIGN KEY (emp_id) REFERENCES employees(emp_id) ON DELETE RESTRICT,
    FOREIGN KEY (sup_id) REFERENCES suppliers(sup_id) ON DELETE RESTRICT
);

-- БИЗНЕС-ЗАДАЧА: "Экран оформления заказа" (п.2 требований к результатам)
-- Бариста выбирает позиции меню — нужно показать их с ценами, 
-- отсортированные по убыванию цены (дорогие сверху)

SELECT 
    m.menu_id,
    m.name AS menu_name,
    m.price
FROM menu m
WHERE m.price > 0
ORDER BY m.price DESC;


-- БИЗНЕС-ЗАДАЧА: "Отчёт по закрытию смены" (п.6 требований)
-- Показать все операции смены с полной детализацией: 
-- кто, что, где, когда, в какой точке

SELECT 
    o.op_id,
    o.op_date,
    p.name AS product_name,
    pt.name AS point_name,
    e.full_name AS employee_name,
    o.op_type,
    o.qty,
    o.op_reason
FROM warehouse_ops o
JOIN shifts s ON o.shift_id = s.shift_id
JOIN points pt ON s.point_id = pt.point_id
JOIN products p ON o.prod_id = p.prod_id
JOIN employees e ON o.emp_id = e.emp_id
WHERE s.shift_id = '00000001'
ORDER BY o.op_date DESC;



-- БИЗНЕС-ЗАДАЧА: "Главный экран смены / Дашборд бариста" (п.1 требований)
-- Метрики вчерашней смены: заработано, чаевые, количество заказов

SELECT 
    SUM(o.qty * m.price) AS total_revenue,     -- общая выручка
    COUNT(DISTINCT o.menu_id) AS orders_count, -- количество заказов
    AVG(m.price) AS avg_check,                 -- средний чек
    MAX(o.qty * m.price) AS max_single_order,  -- самый большой заказ
    MIN(m.price) AS min_item_price,            -- самая дешёвая позиция
    s.tips_sum                                 -- сумма чаевых
FROM warehouse_ops o
JOIN shifts s ON o.shift_id = s.shift_id
JOIN menu m ON o.menu_id = m.menu_id
WHERE o.shift_id = '00000001'
  AND o.op_type = 'Реализация';


-- БИЗНЕС-ЗАДАЧА: "Экран инвентаризации" (п.5 требований)
-- Найти товары, по которым за смену было более 1 операции списания 
-- (это товары, требующие особого внимания при инвентаризации)

SELECT 
    p.name AS product_name,
    COUNT(o.op_id) AS ops_count,
    SUM(o.qty) AS total_qty_moved
FROM warehouse_ops o
JOIN products p ON o.prod_id = p.prod_id
WHERE o.op_type IN ('Списание', 'Списание по рецептуре')
GROUP BY p.prod_id, p.name
HAVING COUNT(o.op_id) > 1
ORDER BY ops_count DESC;



-- БИЗНЕС-ЗАДАЧА: "Получение и регистрация поставки" (п.4 операций)
-- Подзапрос с IN: найти все операции товаров категории "Ингредиенты"
-- (бариста принимает только ингредиенты от поставщиков)
SELECT 
    o.op_id,
    p.name AS product_name,
    o.op_type,
    o.qty,
    o.op_date
FROM warehouse_ops o
JOIN products p ON o.prod_id = p.prod_id
WHERE p.cat_id IN (
    SELECT cat_id FROM categories WHERE name = 'Ингредиенты'
)
ORDER BY o.op_date DESC;

-- Подзапрос с ALL: найти операции прихода, где количество БОЛЬШЕ, 
-- чем количество ЛЮБОЙ операции списания 
-- (т.е. больше максимума списаний — крупные поставки)
SELECT 
    op_id,
    prod_id,
    qty,
    op_date
FROM warehouse_ops
WHERE qty > ALL (
    SELECT qty FROM warehouse_ops 
    WHERE op_type IN ('Списание', 'Списание по рецептуре')
)
AND op_type = 'Приход';

-- Подзапрос с EXISTS: найти сотрудников, у которых были операции 
-- с просроченными товарами (для отчёта руководству)
SELECT 
    e.emp_id,
    e.full_name,
    e.role
FROM employees e
WHERE EXISTS (
    SELECT 1 FROM warehouse_ops o 
    WHERE o.emp_id = e.emp_id 
      AND o.op_reason = 'Просрочка'
);


-- БИЗНЕС-ЗАДАЧА: "Дашборд бариста" — прогресс-бар цели на смену

-- CASE: категоризация операций + оценка выполнения цели смены
SELECT 
    s.shift_id,
    s.shift_target,
    s.cash_balance,
    CASE 
        WHEN s.cash_balance >= s.shift_target THEN '✅ Цель выполнена'
        WHEN s.cash_balance >= s.shift_target * 0.8 THEN '🟡 Почти выполнена'
        WHEN s.cash_balance >= s.shift_target * 0.5 THEN '🟠 В процессе'
        ELSE '🔴 Отстаём от графика'
    END AS target_status,
    ROUND(s.cash_balance / s.shift_target * 100, 1) AS progress_percent
FROM shifts s;

-- CASE в SELECT: понятные подписи для типов операций
SELECT 
    op_id,
    op_type,
    CASE op_type
        WHEN 'Приход' THEN '📥 Поступление от поставщика'
        WHEN 'Списание' THEN CONCAT('🗑️ Списание: ', COALESCE(op_reason, 'без причины'))
        WHEN 'Списание по рецептуре' THEN '☕ Реализация по рецепту'
        WHEN 'Инвентаризация' THEN CONCAT('📋 Инвентаризация: ', COALESCE(op_reason, ''))
        ELSE '🔹 Прочее'
    END AS operation_description,
    qty
FROM warehouse_ops
ORDER BY op_date DESC;

-- IF: направление движения товара (приход/расход)
SELECT 
    op_id,
    prod_id,
    qty,
    IF(op_type = 'Приход', qty, -qty) AS signed_qty,  -- приход "+", расход "-"
    IF(op_type = 'Приход', 'Поступление', 'Расход') AS flow_direction
FROM warehouse_ops;


-- БИЗНЕС-ЗАДАЧА: "Главный экран смены" — готовое представление 
-- с полной информацией об операциях
-- Постоянное представление: сводка всех операций (самый популярный запрос)

CREATE VIEW v_ops_full_info AS
SELECT 
    o.op_id,
    o.op_date,
    o.op_type,
    p.name AS product_name,
    c.name AS category_name,
    pt.name AS point_name,
    e.full_name AS employee_name,
    sl.name AS location_name,
    o.qty,
    o.price_per_unit,
    o.qty * o.price_per_unit AS total_sum,
    o.op_reason,
    o.comment
FROM warehouse_ops o
JOIN products p ON o.prod_id = p.prod_id
JOIN categories c ON p.cat_id = c.cat_id
JOIN shifts s ON o.shift_id = s.shift_id
JOIN points pt ON s.point_id = pt.point_id
JOIN employees e ON o.emp_id = e.emp_id
JOIN storage_locations sl ON o.loc_id = sl.loc_id;


SELECT * FROM v_ops_full_info WHERE op_type = 'Приход';

-- WITH (CTE): расчёт выручки по каждой смене для отчёта закрытия
WITH shift_revenue AS (
    SELECT 
        s.shift_id,
        s.point_id,
        pt.name AS point_name,
        s.open_date,
        s.close_date,
        s.shift_target,
        s.cash_balance,
        s.tips_sum,
        COALESCE(SUM(o.qty * m.price), 0) AS calculated_revenue,
        COUNT(DISTINCT o.menu_id) AS orders_count
    FROM shifts s
    JOIN points pt ON s.point_id = pt.point_id
    LEFT JOIN warehouse_ops o ON s.shift_id = o.shift_id AND o.op_type = 'Реализация'
    LEFT JOIN menu m ON o.menu_id = m.menu_id
    GROUP BY s.shift_id, s.point_id, pt.name, s.open_date, s.close_date, 
             s.shift_target, s.cash_balance, s.tips_sum
)
SELECT 
    shift_id,
    point_name,
    open_date,
    close_date,
    shift_target,
    calculated_revenue,
    cash_balance,
    tips_sum,
    orders_count,
    ROUND(calculated_revenue / NULLIF(shift_target, 0) * 100, 1) AS target_percent
FROM shift_revenue
ORDER BY open_date DESC;
